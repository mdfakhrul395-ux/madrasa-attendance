// ===== এই ব্লকটি firestore.rules এ "match /suggestions/{docId} { ... }" ব্লকের ঠিক নিচে বসান =====

    // ================= প্রশ্ন-উত্তর বক্স =================
    // শিক্ষার্থী শুধু নিজের নামে প্রশ্ন করতে পারে; উত্তর শুধু শিক্ষক লিখতে পারে (answer, answeredBy, answeredAt)
    match /questions/{docId} {
      allow read: if (isTeacherAuth() && resource.data.madrasaId == myMadrasaId()) ||
                     (isStudentAuth() && resource.data.studentId == mySessionStudentId());
      allow create: if isStudentAuth() &&
        request.resource.data.keys().hasOnly(['madrasaId', 'studentId', 'studentName', 'className', 'subject', 'text', 'createdAt']) &&
        request.resource.data.studentId == mySessionStudentId() &&
        request.resource.data.madrasaId == myMadrasaId() &&
        request.resource.data.text is string &&
        request.resource.data.text.size() > 0 &&
        request.resource.data.text.size() <= 1000 &&
        request.resource.data.subject is string && request.resource.data.subject.size() <= 40 &&
        request.resource.data.studentName is string && request.resource.data.studentName.size() <= 80 &&
        request.resource.data.className is string && request.resource.data.className.size() <= 60 &&
        request.resource.data.createdAt is number;
      allow update: if isTeacherAuth() && resource.data.madrasaId == myMadrasaId() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['answer', 'answeredBy', 'answeredAt']) &&
        request.resource.data.answer is string &&
        request.resource.data.answer.size() > 0 &&
        request.resource.data.answer.size() <= 2000;
      allow delete: if (isTeacherAuth() && resource.data.madrasaId == myMadrasaId()) ||
                       (isStudentAuth() && resource.data.studentId == mySessionStudentId() && !('answer' in resource.data));
    }
