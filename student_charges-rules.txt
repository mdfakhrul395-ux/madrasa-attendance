    // ================= শিক্ষার্থী-ভিত্তিক অতিরিক্ত মাসিক চার্জ (student-charges.js) =================
    // গাড়ি ভাড়া, হোস্টেল ইত্যাদি। পড়তে পারেন মাদ্রাসার শিক্ষকগণ; তৈরি/বদলাতে পারেন শুধু অ্যাডমিন (বা সুপার অ্যাডমিন)।
    // মাসের প্রকৃত ফি তৈরি হয় fees_onetime-এ (সেখানকার নিয়ম আগের মতোই)।
    match /student_charges/{chargeId} {
      function chargeShapeOk() {
        return request.resource.data.keys().hasOnly(['madrasaId', 'studentId', 'name', 'amount', 'startMonth', 'active', 'skipMonths', 'createdAt']) &&
          request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 40 &&
          request.resource.data.amount is number && request.resource.data.amount > 0 && request.resource.data.amount <= 1000000 &&
          request.resource.data.startMonth is string && request.resource.data.startMonth.matches('^[0-9]{4}-[0-9]{2}$') &&
          request.resource.data.active is bool &&
          request.resource.data.skipMonths is list && request.resource.data.skipMonths.size() <= 120 &&
          request.resource.data.createdAt is number;
      }

      allow read: if isTeacherAuth() && resource.data.madrasaId == myMadrasaId();

      allow create: if isStudentManager() &&
        request.resource.data.madrasaId == myMadrasaId() &&
        studentInMyMadrasa(request.resource.data.studentId) &&
        chargeShapeOk();

      allow update: if isStudentManager() &&
        resource.data.madrasaId == myMadrasaId() &&
        request.resource.data.madrasaId == resource.data.madrasaId &&
        request.resource.data.studentId == resource.data.studentId &&
        chargeShapeOk();

      allow delete: if false;
    }
